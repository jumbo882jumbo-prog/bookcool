async function loadBooks() {
    try {
        // เรียกใช้ฟังก์ชันที่สร้างไว้ใน supabase-browser.js
        const books = await window.bookcoolDb.getBooks({
            categoryId: 'all',     // หรือระบุ ID เช่น 1
            searchQuery: '',       // ข้อความค้นหา
            sort: 'latest'         // 'latest' | 'price-low' | 'price-high'
        });

        console.log('ข้อมูลหนังสือจาก Supabase:', books);
        // นำ books ไป .map() แสดงลงใน HTML Grid ได้ทันที
        renderBookCards(books);
    } catch (err) {
        console.error('ไม่สามารถดึงข้อมูลได้:', err.message);
    }
}

async function submitOrderToSupabase(cartItems, customerInfo, slipBase64) {
    const orderData = {
        orderId: 'ORD-' + Date.now(),
        userId: auth.getCurrentUser()?.id || null,
        customerName: customerInfo.name,
        customerEmail: customerInfo.email,
        totalAmount: cartItems.reduce((sum, item) => sum + (item.price * item.quantity), 0),
        slipUrl: slipBase64,
        items: cartItems.map(item => ({
            id: item.id,
            title: item.title,
            price: item.price,
            quantity: item.quantity
        }))
    };

    const newOrder = await window.bookcoolDb.createOrder(orderData);
    alert('บันทึกคำสั่งซื้อสำเร็จ! รหัสบิล: ' + newOrder.order_id);
}
3) ตัวอย่างโค้ดสำหรับ React + Vite (supabaseClient.js) ▼ 
// ในโปรเจกต์ React / Vite
import { supabase } from './supabaseClient';
import { useEffect, useState } from 'react';

export function BookList() {
    const [books, setBooks] = useState([]);

    useEffect(() => {
        async function fetchBooks() {
            const { data, error } = await supabase
                .from('ebooks')
                .select('*, categories(category_name)')
                .order('ebook_id', { ascending: false });

            if (!error) setBooks(data);
        }
        fetchBooks();
    }, []);

    return (
        <div className="grid grid-cols-4 gap-4">
            {books.map(b => <div key={b.ebook_id}>{b.title} - ฿{b.price}</div>)}
        </div>
    );
}
